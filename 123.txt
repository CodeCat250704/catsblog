/my-blog-root (项目根目录)
│
├── /modules/                         // 【功能模块区】所有页面均为独立文件夹
│    ├── /home/                       // 首页
│    │    ├── index.html
│    │    ├── style.css
│    │    └── script.js
│    ├── /categories/                 // 分类
│    ├── /collection/                 // 收藏
│    ├── /notice/                     // 通知
│    ├── /co-create/                  // 共创
│    ├── /submit/                     // 投稿（内含文件上传逻辑）
│    ├── /about/                      // 关于
│    └── /settings/                   // 【新增独立文件夹】设置模块
│         ├── index.html              // 设置页面的 UI 结构（如滑块、开关）
│         ├── style.css               // 设置页面特有的 3D 旋钮、玻璃面板样式
│         └── script.js               // 处理主题切换、透明度调节、系统配置等逻辑
│
├── /book/                            // 【核心资源库】支持上传各种格式文件
│    ├── /{文件唯一UUID或Slug}/        // 每一个上传文件，对应一个独立文件夹
│    │    ├── meta.json               // 【核心属性文件】记录标题、作者、上传日期、文件类型
│    │    ├── /md/                    // 存放 Markdown 源文本（用于渲染博客阅读界面）
│    │    │    └── content.md
│    │    ├── /picture/               // 存放该内容的封面图、内嵌图片
│    │    │    ├── cover.jpg
│    │    │    └── illustration.png
│    │    └── /source/                // 【补充】存放原始上传文件（mp3, doc, docx）
│    │         ├── 源文件.docx        // 供用户点击下载
│    │         ├── 背景音乐.mp3       // 供前端调用音频播放器
│    │         └── 其他源文件.md
│    │
│    └── ...（每上传一个内容，自动生成一个新文件夹）
│
├── /assets/                          // 【全局公共资源】
│    ├── /fonts/                      // 自定义字体（建议使用带字重变化的优雅字体）
│    ├── /icons/                      // SVG 矢量图标（包含左侧导航栏所有图标和齿轮图标）
│    ├── common.css                   // 全局统一样式（定义 CSS 变量、毛玻璃核心滤镜）
│    └── common.js                    // 全局交互逻辑（负责裸眼 3D 视差引擎、路由分发）
│
                       // 【主入口文件】承载左侧悬浮导航栏和右侧内容渲染区
